4use crate::algo::collection_ext::CollectionExt;
5use crate::iterators::LazyCollectionIter;
6use crate::{precondition, BidirectionalCollection, LazyCollection};
7
8/// Algorithms for `LazyCollection`.
9pub trait LazyCollectionExt: LazyCollection
10where
11    Self::SubSequence: LazyCollection,
12{
13    /// Returns the "lazily computed" first element.
14    ///
15    /// - Precondition: `!self.is_empty()`.
16    fn lazy_first(&self) -> Self::Element {
17        precondition!(!self.is_empty());
18        self.compute_at(&self.start())
19    }
20
21    /// Returns the "lazily computed" last element.
22    ///
23    /// - Precondition: `!self.is_empty()`.
24    fn lazy_last(&self) -> Self::Element
25    where
26        Self: BidirectionalCollection,
27        Self::SubSequence: BidirectionalCollection,
28    {
29        precondition!(!self.is_empty());
30        self.compute_at(&self.prior(self.end()))
31    }
32
33    /*-----------------Iteration Algorithms-----------------*/
34
35    /// Returns an iterator over the laziily computed elements in collection.
36    fn lazy_iter(&self) -> LazyCollectionIter<'_, Self::SubSequence> {
37        LazyCollectionIter::new(self.full())
38    }
39
40    /// Applies f to each "lazily computed" element of collection.
41    ///
42    /// # Complexity
43    ///   - O(n) where `n == self.count()`.
44    ///
45    /// # Example
46    /// ```rust
47    /// use stl::*;
48    ///
49    /// let arr = 1..4;
50    /// let mut sum = 0;
51    /// arr.lazy_for_each(|x| sum = sum + x);
52    /// assert_eq!(sum, 6);
53    /// ```
54    fn lazy_for_each<F>(&self, mut f: F)
55    where
56        F: FnMut(Self::Element),
57    {
58        let mut start = self.start();
59        let end = self.end();
60        while start != end {
61            f(self.compute_at(&start));
62            start = self.next(start);
63        }
64    }
65
66    /*-----------------Partition Algorithms-----------------*/
67
68    /// Returns two Vec containing the elements of the collection that
69    /// don’t and do satisfy the given predicate, respectively.
70    ///
71    /// # Postcondition
72    ///   - Returns `(falseVec, trueVec)` where `falseVec` contains all elements
73    ///     that don't satisfy predicate and `trueVec` contains all elements
74    ///     that do satisfy predicate.
75    ///   - Relative ordering of elements is preserved in both Vec.
76    ///
77    /// # Complexity
78    ///   - O(n) where `n == self.count()`.
79    ///
80    /// # Examples
81    /// ```rust
82    /// use stl::*;
83    ///
84    /// let arr = 1..=5;
85    /// let (evens, odds) = arr.lazy_partitioned(|x| x % 2 == 1);
86    /// assert_eq!(evens, [2, 4]);
87    /// assert_eq!(odds, [1, 3, 5]);
88    /// ```
89    fn lazy_partitioned<F>(
90        &self,
91        mut belongs_in_second_half: F,
92    ) -> (Vec<Self::Element>, Vec<Self::Element>)
93    where
94        F: FnMut(&Self::Element) -> bool,
95    {
96        let n = self.count();
97        let mut left = Vec::with_capacity(n);
98        let arr = left.spare_capacity_mut();
99        let mut left_idx = 0;
100        let mut right_idx = n;
101
102        for e in self.lazy_iter() {
103            if belongs_in_second_half(&e) {
104                right_idx -= 1;
105                arr[right_idx].write(e);
106            } else {
107                arr[left_idx].write(e);
108                left_idx += 1;
109            }
110        }
111
112        unsafe {
113            left.set_len(n);
114        }
115
116        let mut right = left.split_off(right_idx);
117        right.reverse();
118
119        (left, right)
120    }
121
122    /*-----------------Numeric Algorithms-----------------*/
123
124    /// Returns the result of combining elements of given collection using given
125    /// accumulation operation from left to right.
126    ///
127    /// # Postcondition
128    ///   - Result is `(((init + e1) + e2) + ... + en)`.
129    ///     where e1, e2, ..., en are the lazily computed values of collection,
130    ///     where (a + b) represents op(a, b).
131    ///
132    /// # Complexity:
133    ///   - O(`count`)
134    ///
135    /// # Examples
136    /// ```rust
137    /// use stl::*;
138    ///
139    /// let arr = 1..=3;
140    /// assert_eq!(arr.lazy_fold_left(0, |x, y| x + y), 6);
141    /// ```
142    fn lazy_fold_left<R, F>(&self, init: R, mut op: F) -> R
143    where
144        F: FnMut(R, Self::Element) -> R,
145    {
146        let mut res = init;
147        for e in self.lazy_iter() {
148            res = op(res, e)
149        }
150        res
151    }
152
153    /// Returns the result of combining elements of given collection using given
154    /// accumulation operation from right to left.
155    ///
156    /// # Postcondition
157    ///   - Result is `(e0 + ... + (e(n-1) + (e(n) + init)))`.
158    ///     where e1, e2, ..., en are the references to collection elements,
159    ///     where (a + b) represents op(a, b).
160    ///
161    /// # Complexity:
162    ///   - O(`count`)
163    ///
164    /// # Examples
165    /// ```rust
166    /// use stl::*;
167    ///
168    /// let arr = 1..=3;
169    /// assert_eq!(arr.lazy_fold_right(0, |x, y| x + y), 6);
170    /// ```
171    fn lazy_fold_right<R, F>(&self, init: R, mut op: F) -> R
172    where
173        F: FnMut(Self::Element, R) -> R,
174        Self: BidirectionalCollection,
175        Self::SubSequence: BidirectionalCollection,
176    {
177        let mut res = init;
178        for e in self.lazy_iter().rev() {
179            res = op(e, res)
180        }
181        res
182    }
183}
184
185impl<R> LazyCollectionExt for R
186where
187    R: LazyCollection + ?Sized,
188    R::SubSequence: LazyCollection,
189{
190}
